All captures come from the product's own `--output-format stream-json --verbose` stream
and from the session transcript's `api-request-shape` record. Scratch config dir only.

  "$schema": "https://json-schema.org/draft/2020-12/schema",
  "type": "object",
  "properties": {
   "skill": {
    "description": "The name of a skill from the available-skills list. Do not guess names.",
    "type": "string"
   },
   "args": {

 }
}
